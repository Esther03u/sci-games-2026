# Sci Games Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a full-stack web application for the Sci Games internal sports event — public pages, registration system, live scoring, admin dashboard, and PDF generation.

**Architecture:** Next.js 14 App Router with Supabase (PostgreSQL + Auth + Realtime). Three route groups: `(public)` for spectators, `(admin)` for super admins, `(staff)` for scorekeepers. Server-side API routes handle registration validation and analytics tracking. Supabase Realtime pushes live score updates to all connected clients.

**Tech Stack:** Next.js 14, Vanilla CSS (Glassmorphism), Supabase (PostgreSQL/Auth/Realtime), jsPDF + jspdf-autotable, JSZip, Chart.js

**Spec:** `docs/specs/2026-09-19-sci-games-design.md`

## Global Constraints

- Next.js 14 with App Router (not Pages Router)
- Vanilla CSS only — no Tailwind, no CSS-in-JS
- Thai language UI — use `Sarabun` font for Thai text, `Inter` for headings
- Mobile-first responsive design on all pages
- Glassmorphism design: glass cards with `backdrop-filter: blur(20px)`, gold-amber-orange palette
- All form validation must be enforced server-side (client-side is supplementary only)
- Phone numbers never exposed in public-facing pages or API responses
- Supabase RLS policies enforce access control at database level
- PDPA consent required on registration form
- Every admin mutation must create an audit_log entry

```sql
-- ===== VIEW: team_standings =====
CREATE OR REPLACE VIEW team_standings AS
SELECT
  t.id,
  t.name,
  t.color_hex,
  t.logo_emoji,
  t.sort_order,
  COALESCE(SUM(
    CASE
      WHEN m.team_a_id = t.id THEN m.points_a
      WHEN m.team_b_id = t.id THEN m.points_b
      ELSE 0
    END
  ), 0)::integer AS total_points,
  COUNT(CASE WHEN m.status = 'finished' THEN 1 END)::integer AS matches_played,
  COUNT(CASE WHEN m.status = 'finished'
    AND ((m.team_a_id = t.id AND m.score_a > m.score_b)
      OR (m.team_b_id = t.id AND m.score_b > m.score_a))
    THEN 1 END)::integer AS wins,
  COUNT(CASE WHEN m.status = 'finished'
    AND m.score_a = m.score_b THEN 1 END)::integer AS draws,
  COUNT(CASE WHEN m.status = 'finished'
    AND ((m.team_a_id = t.id AND m.score_a < m.score_b)
      OR (m.team_b_id = t.id AND m.score_b < m.score_a))
    THEN 1 END)::integer AS losses
FROM teams t
LEFT JOIN matches m ON (m.team_a_id = t.id OR m.team_b_id = t.id)
  AND m.status = 'finished'
GROUP BY t.id, t.name, t.color_hex, t.logo_emoji, t.sort_order
ORDER BY total_points DESC, wins DESC;

-- ===== TRIGGER: auto-calculate match points =====
CREATE OR REPLACE FUNCTION calculate_match_points()
RETURNS TRIGGER AS $$
DECLARE
  sport_record RECORD;
BEGIN
  IF NEW.status = 'finished' AND NEW.score_a IS NOT NULL AND NEW.score_b IS NOT NULL THEN
    SELECT win_points, draw_points, lose_points INTO sport_record
    FROM sports WHERE id = NEW.sport_id;

    IF NEW.score_a > NEW.score_b THEN
      NEW.points_a := sport_record.win_points;
      NEW.points_b := sport_record.lose_points;
    ELSIF NEW.score_a < NEW.score_b THEN
      NEW.points_a := sport_record.lose_points;
      NEW.points_b := sport_record.win_points;
    ELSE
      NEW.points_a := sport_record.draw_points;
      NEW.points_b := sport_record.draw_points;
    END IF;
  END IF;

  NEW.updated_at := now();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_match_points
  BEFORE UPDATE ON matches
  FOR EACH ROW
  EXECUTE FUNCTION calculate_match_points();

  NEW.updated_at := now();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;
```

```javascript
'use client';
import { useState, useCallback } from 'react';

export function useToast() {
  const [toasts, setToasts] = useState([]);

  const addToast = useCallback((message, type = 'info', duration = 4000) => {
    const id = Date.now();
    setToasts(prev => [...prev, { id, message, type }]);
    setTimeout(() => {
      setToasts(prev => prev.filter(t => t.id !== id));
    }, duration);
  }, []);

  return { toasts, addToast };
}
```

```javascript
export default function DataTable({ columns, data, searchable = false, searchKeys = [] }) {
  const [search, setSearch] = useState('');
  const [sortCol, setSortCol] = useState(null);
  const [sortDir, setSortDir] = useState('asc');

  const filtered = useMemo(() => {
    if (!search || !searchKeys.length) return data;
    const q = search.toLowerCase();
    return data.filter(row =>
      searchKeys.some(key => String(row[key] || '').toLowerCase().includes(q))
    );
  }, [data, search, searchKeys]);

  const sorted = useMemo(() => {
    if (!sortCol) return filtered;
    return [...filtered].sort((a, b) => {
      const va = a[sortCol] ?? '';
      const vb = b[sortCol] ?? '';
      const cmp = String(va).localeCompare(String(vb), 'th', { numeric: true });
      return sortDir === 'asc' ? cmp : -cmp;
    });
  }, [filtered, sortCol, sortDir]);

  const handleSort = (col) => {
    if (sortCol === col) setSortDir(d => d === 'asc' ? 'desc' : 'asc');
    else { setSortCol(col); setSortDir('asc'); }
  };

  return (
    <div>
      {searchable && (
        <div style={{ marginBottom: '1rem' }}>
          <input
            type="text"
            className="form-input"
            placeholder="🔍 ค้นหา..."
            value={search}
            onChange={e => setSearch(e.target.value)}
            style={{ maxWidth: '320px' }}
          />
        </div>
      )}
      <div style={{ overflowX: 'auto' }}>
        <table className="data-table">
          <thead>
            <tr>
              {columns.map(col => (
                <th
                  key={col.key}
                  onClick={() => col.sortable !== false && handleSort(col.key)}
                  style={{ cursor: col.sortable !== false ? 'pointer' : 'default' }}
                >
                  {col.label}
                  {sortCol === col.key && (sortDir === 'asc' ? ' ▲' : ' ▼')}
                </th>
              ))}
            </tr>
          </thead>
          <tbody>
            {sorted.length === 0 ? (
              <tr><td colSpan={columns.length} style={{ textAlign: 'center', padding: '2rem', color: 'rgba(255,255,255,0.5)' }}>ไม่พบข้อมูล</td></tr>
            ) : sorted.map((row, i) => (

                    {col.render ? col.render(row[col.key], row) : row[col.key]}
                  </td>
                ))}
              </tr>
            ))}
```

```javascript
const rateLimitMap = new Map();

export function rateLimit({ key, limit, windowMs }) {
  const now = Date.now();
  const windowStart = now - windowMs;

  if (!rateLimitMap.has(key)) {
    rateLimitMap.set(key, []);
  }

  const timestamps = rateLimitMap.get(key).filter(ts => ts > windowStart);
  rateLimitMap.set(key, timestamps);

  if (timestamps.length >= limit) {
    return { success: false, remaining: 0 };
  }

  timestamps.push(now);
  return { success: true, remaining: limit - timestamps.length };
}

// Clean up old entries every 5 minutes
if (typeof setInterval !== 'undefined') {
  setInterval(() => {
    const now = Date.now();
    for (const [key, timestamps] of rateLimitMap.entries()) {
      const filtered = timestamps.filter(ts => ts > now - 600000);
      if (filtered.length === 0) rateLimitMap.delete(key);
      else rateLimitMap.set(key, filtered);
    }
  }, 300000);
}
```

```javascript
export function generateVisitorHash(ip, userAgent) {
  // Simple hash — no personal data stored
  const str = `${ip}-${userAgent}`;
  let hash = 0;
  for (let i = 0; i < str.length; i++) {
    const char = str.charCodeAt(i);
    hash = ((hash << 5) - hash) + char;
    hash |= 0;
  }
  return Math.abs(hash).toString(36);
```

```javascript
          supabaseResponse = NextResponse.next({ request });
          cookiesToSet.forEach(({ name, value, options }) =>
            supabaseResponse.cookies.set(name, value, options)
          );
        },
      },
    }
  );

  const { data: { user } } = await supabase.auth.getUser();

  // Protect admin routes
  if (request.nextUrl.pathname.startsWith('/admin') && !request.nextUrl.pathname.startsWith('/admin/login')) {

    if (!user) {
      return NextResponse.redirect(new URL('/staff/login', request.url));
    }
  }

  return supabaseResponse;
}
```

```javascript
import { createAdminClient } from '@/lib/supabase/admin';

export async function validateRegistration({ student_id, full_name, department_id, sport_ids, phone }) {
  const supabase = createAdminClient();
  const errors = [];

  // 1. Student ID format
  if (!/^\d{2}-\d{4}-\d{5}$/.test(student_id)) {
    errors.push({ code: 'INVALID_STUDENT_ID', message: 'รูปแบบรหัสนักศึกษาไม่ถูกต้อง (ตัวอย่าง: 66-1234-12345)' });
    return { valid: false, errors };
  }

  // 2. Name
  if (!full_name || full_name.trim().length < 4) {
    errors.push({ code: 'INVALID_NAME', message: 'กรุณากรอกชื่อ-นามสกุล' });
    return { valid: false, errors };
  }

  // 3. Phone
  if (!/^0[0-9]{8,9}$/.test(phone?.replace(/[-\s]/g, ''))) {
    errors.push({ code: 'INVALID_PHONE', message: 'รูปแบบเบอร์โทรไม่ถูกต้อง' });
    return { valid: false, errors };
  }

  // 4. Duplicate check
  const { data: existing } = await supabase
    .from('athletes')
    .select('id')
    .eq('student_id', student_id)
    .maybeSingle();

  if (existing) {
    errors.push({ code: 'DUPLICATE_REGISTRATION', message: 'รหัสนักศึกษานี้สมัครแล้ว' });
    return { valid: false, errors };
  }

  // 5. Department exists
  const { data: dept } = await supabase
    .from('departments')
    .select('id, team_id')
    .eq('id', department_id)
    .maybeSingle();

  if (!dept) {
    errors.push({ code: 'INVALID_DEPARTMENT', message: 'ไม่พบสาขาวิชาที่เลือก' });
    return { valid: false, errors };
  }

  // 6. Sport count (1-2)
  if (!sport_ids || sport_ids.length < 1 || sport_ids.length > 2) {
    errors.push({ code: 'TOO_MANY_SPORTS', message: 'เลือกกีฬาได้ 1-2 รายการ' });
    return { valid: false, errors };
  }

  // 7. Sports exist
  const { data: sports } = await supabase
    .from('sports')
    .select('id, name, max_players_per_team')
    .in('id', sport_ids);

  if (!sports || sports.length !== sport_ids.length) {
    errors.push({ code: 'INVALID_SPORT', message: 'กีฬาที่เลือกไม่ถูกต้อง' });
    return { valid: false, errors };
  }

  // 8. Quota check per sport per team
  for (const sport of sports) {
    if (sport.max_players_per_team) {
      const { count } = await supabase
        .from('registrations')
        .select('id', { count: 'exact', head: true })
        .eq('sport_id', sport.id)
        .eq('status', 'registered')
        .in('athlete_id',
          (await supabase.from('athletes').select('id').eq('team_id', dept.team_id)).data?.map(a => a.id) || []
        );

      if (count >= sport.max_players_per_team) {
        errors.push({
          code: 'QUOTA_FULL',
          message: `โควตา${sport.name}ของสีนี้เต็มแล้ว (${count}/${sport.max_players_per_team} คน)`
        });
        return { valid: false, errors };
      }
    }
  }

  // 9. Schedule conflict check (only if 2 sports selected)
  if (sport_ids.length === 2) {
    const { data: schedules } = await supabase
      .from('sport_schedules')
      .select('*')
      .in('sport_id', sport_ids);

    const schedA = schedules?.filter(s => s.sport_id === sport_ids[0]) || [];
    const schedB = schedules?.filter(s => s.sport_id === sport_ids[1]) || [];

    for (const a of schedA) {
      for (const b of schedB) {
        if (a.schedule_date === b.schedule_date) {
          if (a.start_time < b.end_time && b.start_time < a.end_time) {
            const sportA = sports.find(s => s.id === sport_ids[0]);
            const sportB = sports.find(s => s.id === sport_ids[1]);
            errors.push({
              code: 'SCHEDULE_CONFLICT',
              message: `ตารางแข่งชนกัน: ${sportA.name} (${a.schedule_date} ${a.start_time}-${a.end_time}) กับ ${sportB.name} (${b.schedule_date} ${b.start_time}-${b.end_time})`
            });
            return { valid: false, errors };
          }
        }
      }
    }
  }

  return { valid: true, errors: [], teamId: dept.team_id };
}
```

```javascript
import { NextResponse } from 'next/server';
import { createAdminClient } from '@/lib/supabase/admin';
import { validateRegistration } from '@/lib/validation';
import { rateLimit } from '@/lib/rate-limit';

export async function POST(request) {
  // Rate limit
  const ip = request.headers.get('x-forwarded-for') || 'unknown';
  const rl = rateLimit({ key: `register:${ip}`, limit: 5, windowMs: 600000 });
  if (!rl.success) {
    return NextResponse.json(
      { success: false, error_code: 'RATE_LIMITED', message: 'ส่งคำขอบ่อยเกินไป กรุณารอสักครู่' },
      { status: 429 }
    );
  }

  const body = await request.json();
  const { student_id, full_name, department_id, sport_ids, phone } = body;

  // Server-side validation
  const result = await validateRegistration({ student_id, full_name, department_id, sport_ids, phone });

  if (!result.valid) {
    return NextResponse.json(
      { success: false, error_code: result.errors[0].code, message: result.errors[0].message },
      { status: 400 }
    );
  }

  // Insert athlete + registrations
  const supabase = createAdminClient();

  const { data: athlete, error: athleteError } = await supabase
    .from('athletes')
    .insert({
      student_id: student_id.trim(),
      full_name: full_name.trim(),
      department_id,
      team_id: result.teamId,
      phone: phone.replace(/[-\s]/g, ''),
    })
    .select('*, departments(name), teams(name, color_hex, logo_emoji)')
    .single();

  if (athleteError) {
    if (athleteError.code === '23505') {
      return NextResponse.json(
        { success: false, error_code: 'DUPLICATE_REGISTRATION', message: 'รหัสนักศึกษานี้สมัครแล้ว' },
        { status: 409 }
      );

      { success: false, error_code: 'SERVER_ERROR', message: 'เกิดข้อผิดพลาด กรุณาลองใหม่' },
      { status: 500 }
    );
  }

  // Insert registrations
  const registrations = sport_ids.map(sid => ({
    athlete_id: athlete.id,
    sport_id: sid,
  }));

  const { error: regError } = await supabase.from('registrations').insert(registrations);

  if (regError) {
    // Rollback athlete
    await supabase.from('athletes').delete().eq('id', athlete.id);
    return NextResponse.json(
      { success: false, error_code: 'SERVER_ERROR', message: 'เกิดข้อผิดพลาดในการลงทะเบียน' },
      { status: 500 }
    );

  return NextResponse.json({
    success: true,
    data: {
      athlete_name: athlete.full_name,
      team_name: athlete.teams.name,
      team_color: athlete.teams.color_hex,
      department: athlete.departments.name,
    }
  });
}
```

```javascript
import { createAdminClient } from '@/lib/supabase/admin';
import { rateLimit } from '@/lib/rate-limit';

export async function POST(request) {
  const ip = request.headers.get('x-forwarded-for') || 'unknown';
  const rl = rateLimit({ key: `check:${ip}`, limit: 10, windowMs: 600000 });
  if (!rl.success) {
    return NextResponse.json(
      { success: false, error_code: 'RATE_LIMITED', message: 'ส่งคำขอบ่อยเกินไป กรุณารอสักครู่' },
      { status: 429 }
    );
  }

  const { student_id, phone } = await request.json();
  const supabase = createAdminClient();

  const { data: athlete } = await supabase
    .from('athletes')
    .select('*, departments(name), teams(name, color_hex, logo_emoji)')
    .eq('student_id', student_id)
    .eq('phone', phone.replace(/[-\s]/g, ''))
    .maybeSingle();

  if (!athlete) {
    return NextResponse.json(
      { success: false, error_code: 'NOT_FOUND', message: 'ไม่พบข้อมูล กรุณาตรวจสอบรหัสนักศึกษาและเบอร์โทร' },
      { status: 404 }
    );
  }

  const { data: registrations } = await supabase
    .from('registrations')
    .select('*, sports(name)')
    .eq('athlete_id', athlete.id);

  return NextResponse.json({
    success: true,
    data: {
      full_name: athlete.full_name,
      student_id: athlete.student_id,
      department: athlete.departments.name,
      team_name: athlete.teams.name,
      team_color: athlete.teams.color_hex,
      team_emoji: athlete.teams.logo_emoji,
      registrations: registrations?.map(r => ({
        sport_name: r.sports.name,
        status: r.status,
        registered_at: r.created_at,
      })) || [],
    }
  });
}
```

```javascript
import jsPDF from 'jspdf';
import 'jspdf-autotable';

export function generateRosterPdf({ sportName, teamName, teamColor, athletes, printDate }) {
  const doc = new jsPDF({ orientation: 'portrait', unit: 'mm', format: 'a4' });

  // Header
  doc.setFontSize(16);
  doc.text('Sci Games - กีฬาสานสัมพันธ์ภายใน', 105, 15, { align: 'center' });
  doc.setFontSize(12);
  doc.text(`กีฬา: ${sportName}`, 14, 30);
  doc.text(`สี: ${teamName}`, 14, 37);
  doc.text(`วันที่พิมพ์: ${printDate}`, 14, 44);

  // Table
  doc.autoTable({
    startY: 52,
    head: [['ลำดับ', 'ชื่อ-นามสกุล', 'รหัสนักศึกษา', 'สาขา', 'ลายเซ็น']],
    body: athletes.map((a, i) => [
      i + 1,
      a.full_name,
      a.student_id,
      a.department_name,
      '', // signature column
    ]),
    styles: { fontSize: 10, cellPadding: 4 },
    headStyles: { fillColor: [39, 39, 42], textColor: 255 },
    columnStyles: { 4: { cellWidth: 35 } },
    alternateRowStyles: { fillColor: [245, 245, 245] },
  });

  return doc;
}
```
